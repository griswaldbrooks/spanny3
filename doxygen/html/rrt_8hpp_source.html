163bool in_collision(like::some_point auto const& p1, like::some_point auto const& p2,
164 std::span<circle_t const> obstacles) {
165 auto const A = distance_squared(p1, p2);
166 return std::ranges::any_of(obstacles, [&](auto const& obstacle) {
167 if (distance_between(p1, obstacle) <= obstacle.radius) {
168 return true;
169 }
170 if (distance_between(p2, obstacle) <= obstacle.radius) {
171 return true;
172 }
173 // Try and solve for quadratic equation that results from finding the intersection between a
174 // line and circle
175 auto const B = 2 * ((p2.x - p1.x) * (p1.x - obstacle.x) + (p2.y - p1.y) * (p1.y - obstacle.y));
176 auto const C = distance_squared(p1, obstacle) - obstacle.radius * obstacle.radius;
177 // If the discriminant is zero or positive, then check if the line segment intersects
178 // as opposed to the infinite line
179 if (auto const discriminant = B * B - 4 * A * C; discriminant >= 0.) {
180 auto const sqrt_d = std::sqrt(discriminant);
181 auto const t1 = (-B + sqrt_d) / (2 * A);
182 auto const t2 = (-B - sqrt_d) / (2 * A);
183 if (is_between(t1, 0., 1.) or is_between(t2, 0., 1.)) {
184 return true;
185 }
186 }
187 // If the discriminant is less zero, then the line does not intersect with the circle
188 return false;
189 });
190}
