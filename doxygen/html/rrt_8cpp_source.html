85std::expected<node_t, std::string> find_neighbor(node_t const& node,
86 std::span<node_t const> nodes) {
87 auto const closest_node = std::ranges::min_element(nodes, [&](auto const& lhs, auto const& rhs) {
88 return distance_between(lhs.position, node.position) <
89 distance_between(rhs.position, node.position);
90 });
91 if (closest_node == nodes.end()) {
92 return std::unexpected(std::string{"Node does not have any neighbors"});
93 }
94 return *closest_node;
95}

97node_t project_sample(planning_context_t const& context, node_t const& sampled,
98 node_t const& closest) {
99 // If the sampled node is too far from the nearest node,
100 // make a closer node in the direction of sampled node
101 if (distance_between(closest.position, sampled.position) > context.sample_distance) {
102 return node_t{project_towards(closest.position, sampled.position, context.sample_distance)};
103 }
104 return sampled;
105}

107std::expected<node_id_t, std::string> expand_tree(planning_context_t const& context,
108 node_t sampled_node, tree_t& tree) {
109 return find_neighbor(sampled_node, tree.nodes)
110 .and_then([&](auto const& closest) -> std::expected<node_id_t, std::string> {
111 auto const sample = project_sample(context, sampled_node, closest);
112 if (in_collision(closest.position, sample.position, context.obstacles)) {
113 return std::unexpected(std::string{"Sampled node was in collision"});
114 }
115 // Add to tree
116 auto const cost = heuristic(closest, sample);
117 tree.nodes.emplace_back(sample.position);
118 tree.edges.emplace_back(closest.id, sample.id, cost);
119 return sample.id;
120 });
121}
